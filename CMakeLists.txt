cmake_minimum_required(VERSION 3.16)
project(ModeratedLMS VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
include(CTest)
add_library(lms_core src/core.cpp)
target_include_directories(lms_core PUBLIC include)
target_link_libraries(lms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
if(MSVC)
 target_compile_options(lms_core PRIVATE /W4)
else()
 target_compile_options(lms_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(lms src/main.cpp)
target_link_libraries(lms PRIVATE lms_core)
add_executable(lms_tests tests/test_core.cpp)
target_link_libraries(lms_tests PRIVATE lms_core)
add_test(NAME LMS_Unit_Integration_Tests COMMAND lms_tests)
